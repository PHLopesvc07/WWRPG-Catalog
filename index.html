⚙️ Conexão Firebase Firestore (`wwcatalog-10546`)

1. Regras de Segurança do Firestore (`firestore.rules`): Cole as regras abaixo na aba Regras do Console do Firebase e clique em Publicar para habilitar a leitura e gravação das 4 coleções (creatures, spells, plants, potions e _indexes):

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isValidCatalogWrite() {
      return request.resource.data.keys().size() > 0
             && request.resource.data.keys().size() <= 40;
    }

    match /creatures/{creatureId} {
      allow read: if true;
      allow create, update: if isValidCatalogWrite() && (request.resource.data.nome is string);
      allow delete: if true;
    }

    match /spells/{spellId} {
      allow read: if true;
      allow create, update: if isValidCatalogWrite() && (request.resource.data.name is string);
      allow delete: if true;
    }

    match /plants/{plantId} {
      allow read: if true;
      allow create, update: if isValidCatalogWrite() && (request.resource.data.nome is string);
      allow delete: if true;
    }

    match /potions/{potionId} {
      allow read: if true;
      allow create, update: if isValidCatalogWrite() && (request.resource.data.nome is string);
      allow delete: if true;
    }

    match /_indexes/{collectionName} {
      allow read, write: if true;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
